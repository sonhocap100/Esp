local ESP = {}

-- --- HÀM KHỞI TẠO ĐỐI TƯỢNG VẼ ---
local function CreateLine()
    local line = Drawing.new("Line")
    line.Thickness = 1.5; line.Transparency = 1; line.Visible = false
    return line
end

local function AddESP(player)
    if player == LP or ESP[player] then return end
    
    local objects = {
        BoxLines = {}, 
        SkeletonLines = {},
        Tracer = CreateLine(), 
        Name = Drawing.new("Text")
    }
    
    for i = 1, 12 do table.insert(objects.BoxLines, CreateLine()) end
    for i = 1, 15 do table.insert(objects.SkeletonLines, CreateLine()) end
    
    objects.Name.Size = 14; objects.Name.Center = true; objects.Name.Outline = true
    ESP[player] = objects
end

local function RemoveESP(player)
    if ESP[player] then
        for _, v in pairs(ESP[player].BoxLines) do v:Remove() end
        for _, v in pairs(ESP[player].SkeletonLines) do v:Remove() end
        ESP[player].Tracer:Remove(); ESP[player].Name:Remove(); ESP[player] = nil
    end
end

local function HideAll(obj)
    for _, l in pairs(obj.BoxLines) do l.Visible = false end
    for _, l in pairs(obj.SkeletonLines) do l.Visible = false end
    obj.Tracer.Visible = false; obj.Name.Visible = false
end

local function GetCharacter(player)
    if player.Character and player.Character.Parent then return player.Character end
    for _, v in pairs(workspace:GetChildren()) do
        if v.Name == player.Name and v:IsA("Model") then return v end
    end
    return nil
end

-- --- VÒNG LẶP RENDER CHÍNH ---
RunService.RenderStepped:Connect(function()
    for _, player in pairs(Players:GetPlayers()) do
        if not ESP[player] then AddESP(player) continue end
        local obj = ESP[player]
        
        local char = GetCharacter(player)
        local hrp = char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChildWhichIsA("BasePart"))
        
        -- [MỚI] KIỂM TRA ĐỒNG ĐỘI: Nếu bật TeamCheck và người chơi chung đội với bạn -> Ẩn ESP
        local isTeamate = Settings.TeamCheck and player.Team == LP.Team
        
        if not Settings.Enabled or not char or not hrp or isTeamate then
            HideAll(obj)
            continue
        end

        local dist = (Camera.CFrame.Position - hrp.Position).Magnitude
        local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
        
        if not onScreen or dist > Settings.MaxDistance then
            HideAll(obj)
            continue
        end

        -- --- 1. VẼ HÌNH HỘP 3D ---
        if Settings.Box3D then
            local size = Vector3.new(2, 3, 1.2)
            local cf = hrp.CFrame
            local points = {
                Camera:WorldToViewportPoint((cf * CFrame.new(size.X, size.Y, size.Z)).Position),
                Camera:WorldToViewportPoint((cf * CFrame.new(-size.X, size.Y, size.Z)).Position),
                Camera:WorldToViewportPoint((cf * CFrame.new(-size.X, -size.Y, size.Z)).Position),
                Camera:WorldToViewportPoint((cf * CFrame.new(size.X, -size.Y, size.Z)).Position),
                Camera:WorldToViewportPoint((cf * CFrame.new(size.X, size.Y, -size.Z)).Position),
                Camera:WorldToViewportPoint((cf * CFrame.new(-size.X, size.Y, -size.Z)).Position),
                Camera:WorldToViewportPoint((cf * CFrame.new(-size.X, -size.Y, -size.Z)).Position),
                Camera:WorldToViewportPoint((cf * CFrame.new(size.X, -size.Y, -size.Z)).Position)
            }
            local edges = {{1,2},{2,3},{3,4},{4,1},{5,6},{6,7},{7,8},{8,5},{1,5},{2,6},{3,7},{4,8}}
            for i, edge in pairs(edges) do
                local l = obj.BoxLines[i]
                l.From = Vector2.new(points[edge[1]].X, points[edge[1]].Y)
                l.To = Vector2.new(points[edge[2]].X, points[edge[2]].Y)
                l.Color = Settings.Color; l.Visible = true
            end
        else for _, l in pairs(obj.BoxLines) do l.Visible = false end end

        -- --- 2. VẼ KHUNG XƯƠNG ---
        if Settings.Skeleton and dist < 1000 then 
            local function getV2(partName)
                local p = char:FindFirstChild(partName)
                if p then 
                    local v, vis = Camera:WorldToViewportPoint(p.Position)
                    return Vector2.new(v.X, v.Y), vis
                end
                return nil, false
            end
            local joints = {
                {"Head", "UpperTorso"}, {"UpperTorso", "LowerTorso"},
                {"Head", "Torso"}, {"Torso", "Left Arm"}, {"Torso", "Right Arm"},
                {"Torso", "Left Leg"}, {"Torso", "Right Leg"},
                {"LowerTorso", "LeftUpperLeg"}, {"LeftUpperLeg", "LeftLowerLeg"},
                {"LowerTorso", "RightUpperLeg"}, {"RightUpperLeg", "RightLowerLeg"}
            }
            for i, joint in pairs(joints) do
                local p1, vis1 = getV2(joint[1]); local p2, vis2 = getV2(joint[2])
                local l = obj.SkeletonLines[i]
                if l and p1 and p2 and (vis1 or vis2) then
                    l.From = p1; l.To = p2; l.Color = Settings.Color; l.Visible = true
                elseif l then l.Visible = false end
            end
        else for _, l in pairs(obj.SkeletonLines) do l.Visible = false end end

        -- --- 3. VẼ TRACER ---
        if Settings.Tracer then
            local s = Camera.ViewportSize
            local o = Settings.TracerOrigin == "Top" and Vector2.new(s.X/2, 0) or Settings.TracerOrigin == "Center" and Vector2.new(s.X/2, s.Y/2) or Vector2.new(s.X/2, s.Y)
            obj.Tracer.From = o; obj.Tracer.To = Vector2.new(pos.X, pos.Y); obj.Tracer.Color = Settings.Color; obj.Tracer.Visible = true
        else obj.Tracer.Visible = false end

        -- --- 4. TÊN & KHOẢNG CÁCH ---
        if Settings.Name or Settings.Distance then
            local t = (Settings.Name and player.Name or "") .. (Settings.Distance and " [" .. math.floor(dist) .. "m]" or "")
            obj.Name.Text = t; obj.Name.Position = Vector2.new(pos.X, pos.Y - 60); obj.Name.Color = Settings.Color; obj.Name.Visible = true
        else obj.Name.Visible = false end
    end
end)

local function MakeDraggable(guiObject)
    local dragging, dragInput, dragStart, startPos
    guiObject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = guiObject.Position
            input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
        end
    end)
    guiObject.InputChanged:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end end)
    UserInputService.InputChanged:Connect(function(input) if input == dragInput and dragging then
        local delta = input.Position - dragStart
        guiObject.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end end)
end
MakeDraggable(MainFrame); MakeDraggable(ToggleBtn)

end)

local function createBtn(label, settingKey, callback)
    local b = Instance.new("TextButton", Scroll)
    b.Size = UDim2.new(0.9, 0, 0, 35); b.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    b.TextColor3 = Color3.new(1, 1, 1); b.Text = label; Instance.new("UICorner", b)
    local function update()
        if settingKey and typeof(Settings[settingKey]) == "boolean" then
            b.Text = label .. ": " .. (Settings[settingKey] and "ON" or "OFF")
            b.TextColor3 = Settings[settingKey] and Color3.new(0, 1, 0.5) or Color3.new(1, 0.3, 0.3)
        elseif settingKey == "TracerOrigin" then b.Text = "GỐC: " .. Settings.TracerOrigin
        else b.Text = label end
    end
    b.MouseButton1Click:Connect(function() if callback then callback(b) elseif settingKey then Settings[settingKey] = not Settings[settingKey] end update() end)
    update(); return b
end
